Voice Bot System Architecture & Implementation Blueprint

How the 3 core subsystems (Priority Scroll Arbiter, Continuous Wake-Word State Machine, and PDF Text-to-Teacher Engine) work together.

1. Priority Scroll Arbiter

Separates programmatic bot scrolls (isBotScrolling = true) from human input events (wheel, touchmove, keydown). Any human scroll immediately sets scrollMode = 'USER_OVERRIDE' and starts a configurable cooldown timer so the bot never fights your hand on the trackpad or mouse wheel.

2. Active Listening State Machine

Runs continuous webkitSpeechRecognition with interimResults = true for sub-second keyword interception. Matching "doubt" cancels TTS and captures your question; "give me a minute" pauses TTS & scroll; "continue" resumes teaching from the active block.

3. PDF.js Block & Teacher Sync

Extracts text items with Y-coordinates via page.getTextContent(), groups them into logical teaching paragraphs, and links each block to a DOM node so the teacher highlights, explains, and scrolls block-by-block.

// 1. PRIORITY SCROLL ARBITER (Human Always Beats Bot)
class PriorityScrollArbiter {
  constructor(viewportEl, cooldownMs = 6000) {
    this.viewport = viewportEl;
    this.cooldownMs = cooldownMs;
    this.userHasPriority = false;
    this.isProgrammaticScroll = false;
    this.cooldownTimer = null;

    const onHumanScrollInput = () => {
      if (this.isProgrammaticScroll) return;
      this.userHasPriority = true;
      clearTimeout(this.cooldownTimer);
      this.cooldownTimer = setTimeout(() => {
        this.userHasPriority = false;
      }, this.cooldownMs);
    };

    ['wheel', 'touchstart', 'touchmove', 'mousedown'].forEach(evt =>
      this.viewport.addEventListener(evt, onHumanScrollInput, { passive: true })
    );
  }

  botScrollToElement(targetEl) {
    if (this.userHasPriority || !targetEl) return false; // Yield to human!
    this.isProgrammaticScroll = true;
    targetEl.scrollIntoView({ behavior: 'smooth', block: 'center' });
    setTimeout(() => { this.isProgrammaticScroll = false; }, 650);
    return true;
  }
}

// 2. ACTIVE LISTENING WAKE-WORD ROUTER ("doubt", "give me a minute", "continue")
function evaluateVoiceTranscript(transcript, botState, actions) {
  const clean = transcript.toLowerCase().trim();
  if (/\b(doubt|wait i have a doubt|hold on i have a doubt|question)\b/.test(clean)) {
    actions.enterDoubtMode();
    return 'DOUBT_MODE';
  }
  if (/\b(give me a minute|hold on|pause|wait a second|one minute)\b/.test(clean)) {
    actions.enterPausedMode();
    return 'PAUSED_WAITING';
  }
  if (/\b(continue|go ahead|resume|keep going|carry on)\b/.test(clean)) {
    actions.resumeTeaching();
    return 'TEACHING';
  }
  if (botState === 'DOUBT_MODE' && clean.length > 8) {
    actions.answerStudentDoubt(transcript);
  }
  return botState;
}
